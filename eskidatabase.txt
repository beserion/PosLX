 CREATE PROCEDURE [dbo].[KasaRaporu] 	@StartDate Date, 	@EndDate Date AS BEGIN 	SET NOCOUNT ON;     SELECT 	K.KasaAdi, 	K.Bakiye + ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'KG' and KasaNo=k.KasaNo and Tarih > K.KapanisTarihi),0) AS Giris, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'KC' and KasaNo=k.KasaNo and Tarih > K.KapanisTarihi),0) AS Cikis, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'SA' and KasaNo=k.KasaNo AND OdemeTuru=1 and Tarih > K.KapanisTarihi),0) AS Nakit, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'SA' and KasaNo=k.KasaNo AND OdemeTuru=2 and Tarih > K.KapanisTarihi),0) AS KrediKarti, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'SA' and KasaNo=k.KasaNo AND OdemeTuru in (3,4) and Tarih > K.KapanisTarihi),0) AS CekSenet, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'SA' and KasaNo=k.KasaNo AND OdemeTuru =5 and Tarih > K.KapanisTarihi),0) AS Veresiye, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'TH' and KasaNo=k.KasaNo and Barkod in (select Barkot from Musteri) and Tarih > K.KapanisTarihi),0) AS Tahsilat, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'TH' and KasaNo=k.KasaNo and Barkod in (select Barkot from Toptanci) and Tarih > K.KapanisTarihi),0) AS Odeme, 	ISNULL((SELECT SUM(TOPLAM) FROM HareketBaslik WHERE HareketTuru = 'GI' and KasaNo=k.KasaNo and Tarih > K.KapanisTarihi),0) AS Gider FROM  	Kasa K END

 CREATE PROCEDURE [dbo].[SearchUrun] 	 	@Query nvarchar(255) AS BEGIN 	 	 	SET NOCOUNT ON;      	select top 50 Adi from Urun where  Adi like N'%' + replace(@Query,' ','%') + '%' and aktif=1 order by Adi asc END

 CREATE PROCEDURE [dbo].[UrunHareketTarih] 	 	@StartDate Date, 	 	@EndDate Date  AS BEGIN 	 	SET NOCOUNT ON; 	 	SET @EndDate = Dateadd(day,1,@EndDate); 	 	select  		u.UrunId,         		u.Barkot, 	    		u.Adi, 	    		ug.Isim , 	    		case Birim when 1 then 'Adet' when 2 then 'KG' else 'Metre' end as Birim, 	    		case Birim when 1 then StokAdedi * 1.00 else StokAdedi / 1000.00 end as Bakiye, 	    		case Birim when 1 then SatisFiyati * StokAdedi * 1.00 else SatisFiyati * StokAdedi / 1000.00 end as BakiyeDeger, 	    		isnull(hd.Adet,0) as DevirMiktar, 	    		isnull(hd.Tutar,0) as DevirTutar, 	    		isnull(ha.Adet,0) as AlimMiktar, 	    		isnull(ha.Tutar,0) as AlimTutar, 	    		isnull(hs.Adet,0) as SatisMiktar, 	    		isnull(hs.Tutar,0) as SatisTutar, 	    		isnull(hs.Kar,0) as Kar  	from Urun u  		inner join UrunGrup ug on ug.Id = u.GrupId  		left join ( 	 			select h.Barkot, 	        			sum(case h.Birim when 1 then h.Adet * 1.00 else h.Adet /1000.00 end) as Adet, 			sum(case h.Birim when 1 then h.Adet * h.SatisFiyati * 1.00 else h.Adet * h.SatisFiyati / 1000.00 end) as Tutar 			from Hareket h 	join HareketBaslik hb on h.HareketNo = hb.HareketNo 	 			where hb.Tarih between @StartDate and @EndDate and h.HareketTuru='DE' 	group by h.Barkot  		) hd on hd.Barkot = u.Barkot  		left join ( 	 			  select h.Barkot, 	        			  sum(case h.Birim when 1 then h.Adet * 1.00 else h.Adet /1000.00 end) as Adet, 			  sum(case h.Birim when 1 then h.Adet * h.BirimFiyat * 1.00 else h.Adet * h.BirimFiyat / 1000.00 end) as Tutar 	 			  from AlisFaturasiUrun h 	join AlisFaturasi hb on h.FaturaID = hb.ID 	 			  where hb.Tarih between @StartDate and @EndDate group by h.Barkot  		  ) ha on ha.Barkot = u.Barkot  		left join ( 	 			select h.Barkot, 	        			sum(case h.Birim when 1 then h.Adet * 1.00 else h.Adet /1000.00 end) as Adet , 			sum(case h.Birim when 1 then h.Adet * h.SatisFiyati * 1.00 else h.Adet * h.SatisFiyati / 1000.00 end) as Tutar, 			sum(case h.Birim when 1 then h.Adet * (h.SatisFiyati - (h.AlisFiyati*(1+(h.KDV/100.00)))) * 1.00 else h.Adet * (h.SatisFiyati - (h.AlisFiyati*(1+(h.KDV/100.00)))) / 1000.00 end) as Kar		 			from Hareket h 	join HareketBaslik hb on h.HareketNo = hb.HareketNo 	 			where hb.Tarih between @StartDate and @EndDate and h.HareketTuru='SA' 	group by h.Barkot  		) hs on hs.Barkot = u.Barkot where u.Aktif = 1  END
GO
